CREATE TABLE Proposal (
    proposalID INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    agency VARCHAR(255) NOT NULL,
    otherAgencyName VARCHAR(255),
    status ENUM('Submitted', 'Accepted', 'Granted') NOT NULL,
    amountClaimed DECIMAL(15, 2) NOT NULL CHECK (amountClaimed >= 0),
    PI VARCHAR(255) NOT NULL,
    CoPI VARCHAR(255) NOT NULL,
    dateOfSubmission DATE NOT NULL,
    proposalPDF LONGBLOB,
    createdAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    userId INT NOT NULL,
    FOREIGN KEY (userId) REFERENCES User(userId) ON DELETE CASCADE ON UPDATE CASCADE
)
